65  void computeError() override {
66  HATEB_ASSERT_MSG(cfg_, "You must call setParameters() on EdgeAgentRobotVisibility()");
67  const auto* robot_bandpt = static_cast<const VertexPose*>(_vertices[0]);
68  const auto* agent_bandpt = static_cast<const VertexPose*>(_vertices[1]);
69 
70  Eigen::Vector2d d_rtoh = agent_bandpt->position() - robot_bandpt->position();
71  Eigen::Vector2d d_htor = robot_bandpt->position() - agent_bandpt->position();
72  Eigen::Vector2d agent_look_at = {cos(agent_bandpt->theta()), sin(agent_bandpt->theta())};
73  Eigen::Vector2d robot_look_at = {cos(robot_bandpt->theta()), sin(robot_bandpt->theta())};
74 
75  // calculate the angle between human heading and direction vector from human to robot
76  double delta_psi = fabs(acos(agent_look_at.dot(d_htor) / (agent_look_at.norm() * d_htor.norm())));
77  double c_visibility = 0.0;
78  // Dot product of headings
79  double ang = agent_look_at.dot(robot_look_at);
80 
81  // Apply the cost if the robot is not within the FOV of human
82  if (delta_psi >= cfg_->agent.fov * M_PI / 180) {
83  if (ang >= 0) {
84  c_visibility = 5 * ((std::pow(2, -(std::pow(d_rtoh.x(), 2)))) * (std::pow(2, -(std::pow(d_rtoh.y(), 2)))));
85  } else {
86  c_visibility = 0.;
87  }
88  }
89 
90  _error[0] = penaltyBoundFromAbove(c_visibility, cfg_->hateb.visibility_cost_threshold, cfg_->optim.penalty_epsilon);
91 
92  HATEB_ASSERT_MSG(std::isfinite(_error[0]), "EdgeAgentRobotVisibility::computeError() _error[0]=%f\n", _error[0]);
93  }
