181  void startWorker(rclcpp::Rate::SharedPtr rate, nav2_costmap_2d::Costmap2D* costmap, bool spin_thread = false)
182  {
183  setCostmap2D(costmap);
184 
185  if (spin_thread_)
186  {
187  {
188  std::lock_guard<std::mutex> terminate_lock(terminate_mutex_);
189  need_to_terminate_ = true;
190  }
191  spin_thread_->join();
192  delete spin_thread_;
193  }
194 
195  if (spin_thread)
196  {
197  RCLCPP_DEBUG(nh_->get_logger(), "costmap_converter", "Spinning up a thread for the CostmapToPolygons plugin");
198  need_to_terminate_ = false;
199 
200  worker_timer_ = nh_->create_wall_timer(
201  rate->period(),
202  std::bind(&BaseCostmapToPolygons::workerCallback, this));
203  spin_thread_ = new std::thread(std::bind(&BaseCostmapToPolygons::spinThread, this));
204  }
205  else
206  {
207  worker_timer_ = nh_->create_wall_timer(
208  rate->period(),
209  std::bind(&BaseCostmapToPolygons::workerCallback, this));
210  spin_thread_ = nullptr;
211  }
212  }

228  }
