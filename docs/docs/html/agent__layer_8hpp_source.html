149  static double Gaussian2D(double x, double y, double x0, double y0, double A, double varx, double vary) {
150  double dx = x - x0;
151  double dy = y - y0;
152  double d = sqrt((dx * dx) + (dy * dy));
153  double theta = atan2(dy, dx);
154  double xx = d * cos(theta);
155  double yy = d * sin(theta);
156  return A / std::max(d, 1.0) * Guassian1D(xx, 0.0, 1.0, varx) * Guassian1D(yy, 0.0, 1.0, vary);
157  }

171  static double Gaussian2D_skewed(double x, double y, double x0, double y0, double A, double varx, double vary, double skew_ang) {
172  double dx = x - x0;
173  double dy = y - y0;
174  double d = sqrt((dx * dx) + (dy * dy));
175  double theta = atan2(dy, dx);
176  double xx = d * cos(theta - skew_ang);
177  double yy = d * sin(theta - skew_ang);
178  return A / std::max(d, 1.0) * Guassian1D(xx, 0.0, 1.0, varx) * Guassian1D(yy, 0.0, 1.0, vary);
179  }
