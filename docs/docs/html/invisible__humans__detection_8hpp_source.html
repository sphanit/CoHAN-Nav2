130  bool worldToMap(double wx, double wy, int& mx, int& my) const {
131  if (wx < origin_x_ || wy < origin_y_) return false;
132 
133  mx = static_cast<int>((wx - origin_x_) / resolution_);
134  my = static_cast<int>((wy - origin_y_) / resolution_);
135 
136  return mx < size_x_ && my < size_y_;
137  }

155  static Point getLeftPoint(Point p1, Point p2, Point p3, double dist = 1) {
156  double x = p2.first - p1.first;
157  double y = p2.second - p1.second;
158  double point_dist = std::hypot(x, y);
159 
160  Point p;
161  p.first = p3.first - (dist * y / point_dist);
162  p.second = p3.second + (dist * x / point_dist);
163  return p;
164  }
165 
174  static Point getRightPoint(Point p1, Point p2, Point p3, double dist = 1) {
175  double x = p2.first - p1.first;
176  double y = p2.second - p1.second;
177  double point_dist = std::hypot(x, y);
178 
179  Point p;
180  p.first = p3.first + (dist * y / point_dist);
181  p.second = p3.second - (dist * x / point_dist);
182  return p;
183  }

192  static std::vector<Point> getTwoPoints(Point p1, Point p2, double radius) {
193  std::vector<Point> points;
194  auto l_p = getLeftPoint(p1, p2, p2, radius);
195  points.push_back(l_p);
196  auto r_p = getRightPoint(p1, p2, p2, radius);
197  points.push_back(r_p);
