72  const auto* agent_bandpt_nxt = static_cast<const VertexPose*>(_vertices[4]);
73  const auto* dt_agent = static_cast<const VertexTimeDiff*>(_vertices[5]);
74 
75  // Get the velocities of the robot and agent
76  Eigen::Vector2d diff_robot = robot_bandpt_nxt->position() - robot_bandpt->position();
77  Eigen::Vector2d robot_vel = diff_robot / dt_robot->dt();
78  Eigen::Vector2d diff_agent = agent_bandpt_nxt->position() - agent_bandpt->position();
79  Eigen::Vector2d agent_vel = diff_agent / dt_agent->dt();
80

83  Eigen::Vector2d d_htor = robot_bandpt->position() - agent_bandpt->position();
84 
85  // Cost definition
86  double rel_vel_cost = (std::max((robot_vel - agent_vel).dot(d_rtoh), 0.0) + robot_vel.norm() + 1) / d_rtoh.norm();
87 
88  _error[0] = penaltyBoundFromAbove(rel_vel_cost, cfg_->hateb.rel_vel_cost_threshold, cfg_->optim.penalty_epsilon);
