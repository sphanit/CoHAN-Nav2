75  void computeError() override {
76  HATEB_ASSERT_MSG(cfg_ && _measurement, "You must call setHATebConfig() on EdgeInvisibleHuman()");
77  const auto* bandpt = static_cast<const VertexPose*>(_vertices[0]);
78  const auto* bandpt_nxt = static_cast<const VertexPose*>(_vertices[1]);
79  const auto* dt = static_cast<const VertexTimeDiff*>(_vertices[2]);
80 
81  // Get the distance and the velocity
82  double dist = cfg_->robot_model->calculateDistance(bandpt->pose(), _measurement);
83  auto robot_vel = (bandpt_nxt->position() - bandpt->position()) / dt->dt();
84 
85  double cost = dist / (V_i_ + robot_vel.norm() - 0.1);
86  if (t_ > 0.5) { // Accounting for human reaction time
87  // Cost calculation
88  cost = dist / (std::max(V_i_ - (a_norm_ * t_) + robot_vel.norm() - 0.1, 0.01));
89  }
90  _error[0] = penaltyBoundFromBelow(cost, cfg_->hateb.invisible_human_threshold, cfg_->optim.penalty_epsilon);
91 
92  HATEB_ASSERT_MSG(std::isfinite(_error[0]), "EdgeInvisibleHuman::computeError() _error[0]=%f\n", _error[0]);
93  }
