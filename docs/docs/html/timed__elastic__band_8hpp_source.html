588 bool TimedElasticBand::initTrajectoryToGoal(BidirIter path_start, BidirIter path_end, Fun fun_position, double max_vel_x, double max_vel_theta, boost::optional<double> max_acc_x,
589  boost::optional<double> max_acc_theta, boost::optional<double> start_orientation, boost::optional<double> goal_orientation, int min_samples,
590  bool guess_backwards_motion) {
591  Eigen::Vector2d start_position = fun_position(*path_start);
592  Eigen::Vector2d goal_position = fun_position(*std::prev(path_end));
593 
594  bool backwards = false;
595 
596  double start_orient;
597  double goal_orient;
598  if (start_orientation) {
599  start_orient = *start_orientation;
600 
601  // check if the goal is behind the start pose (w.r.t. start orientation)
602  if (guess_backwards_motion && (goal_position - start_position).dot(Eigen::Vector2d(std::cos(start_orient), std::sin(start_orient))) < 0) {
603  backwards = true;
604  }
605  } else {
606  Eigen::Vector2d start2goal = goal_position - start_position;
607  start_orient = atan2(start2goal[1], start2goal[0]);
608  }
609 
610  double timestep = 1; // TODO: time
611 
612  if (goal_orientation) {
613  goal_orient = *goal_orientation;
614  } else {
615  goal_orient = start_orient;
616  }
617 
618  if (!isInit()) {
619  addPose(start_position, start_orient, true); // add starting point and mark it as fixed for optimization
620 
621  // we insert middle points now (increase start by 1 and decrease goal by 1)
622  std::advance(path_start, 1);
623  std::advance(path_end, -1);
624  int idx = 0;
625  for (; path_start != path_end; ++path_start) // insert middle-points
626  {
627  // Eigen::Vector2d point_to_goal = path.back()-*it;
628  // double dir_to_goal = atan2(point_to_goal[1],point_to_goal[0]); // direction to goal
629  // Alternative: Direction from last path
630  Eigen::Vector2d curr_point = fun_position(*path_start);
631  Eigen::Vector2d diff_last = curr_point - Pose(idx).position(); // we do not use boost::prior(*path_start) for those cases,
632  // where fun_position() does not return a reference or is expensive.
633  double diff_norm = diff_last.norm();
634 
635  double timestep_vel = diff_norm / max_vel_x; // constant velocity
636  double timestep_acc;
637 
638  if (max_acc_x) {
639  timestep_acc = sqrt(2 * diff_norm / (*max_acc_x)); // constant acceleration
640  if (timestep_vel < timestep_acc && max_acc_x) {
641  timestep = timestep_acc;
642  } else {
643  timestep = timestep_vel;
644  }
645  } else {
646  timestep = timestep_vel;
647  }
648 
649  if (timestep <= 0) {
650  timestep = 0.2; // TODO: this is an assumption
651  }
652 
653  double yaw = atan2(diff_last[1], diff_last[0]);
654  if (backwards) {
655  yaw = g2o::normalize_theta(yaw + M_PI);
656  }
657  addPoseAndTimeDiff(curr_point, yaw, timestep);
658 
659  ++idx;
660  }
661  Eigen::Vector2d diff = goal_position - Pose(idx).position();
662  double diff_norm = diff.norm();
663  double timestep_vel = diff_norm / max_vel_x; // constant velocity
664  if (max_acc_x) {
665  double timestep_acc = sqrt(2 * diff_norm / (*max_acc_x)); // constant acceleration
666  if (timestep_vel < timestep_acc) {
667  timestep = timestep_acc;
668  } else {
669  timestep = timestep_vel;
670  }
671  } else {
672  timestep = timestep_vel;
673  }
674 
675  PoseSE2 goal(goal_position, goal_orient);
676 
677  // if number of samples is not larger than min_samples, insert manually
678  if (sizePoses() < min_samples - 1) {
679  RCLCPP_DEBUG(rclcpp::get_logger("TimedElasticBand"), "initTEBtoGoal(): number of generated samples is less than specified by min_samples. Forcing the insertion of more samples...");
680  while (sizePoses() < min_samples - 1) // subtract goal point that will be added later
681  {
682  // Each inserted point bisects the remaining distance. Thus the timestep is also bisected.
683  timestep /= 2;
684  // simple strategy: interpolate between the current pose and the goal
685  addPoseAndTimeDiff(PoseSE2::average(BackPose(), goal), timestep); // let the optimier correct the timestep (TODO: better initialization
686  }
687  }
688 
689  // now add goal
690  addPoseAndTimeDiff(goal, timestep); // add goal point
691  setPoseVertexFixed(sizePoses() - 1, true); // GoalConf is a fixed constraint during optimization
692  } else // size!=0
693  {
694  RCLCPP_WARN(rclcpp::get_logger("TimedElasticBand"),
695  "Cannot init TEB between given configuration and goal, because TEB vectors are not empty or TEB is already initialized (call this function before adding states yourself)!");
696  RCLCPP_WARN(rclcpp::get_logger("TimedElasticBand"), "Number of TEB configurations: %d, Number of TEB timediffs: %d", sizePoses(), sizeTimeDiffs());
697  return false;
698  }
699  return true;
700 }
