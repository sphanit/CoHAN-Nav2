264  void loadBoundParameters() {
265  for (const auto& [name, updater] : param_bindings_) {
266  try {
267  rclcpp::Parameter param;
268  if (is_lifecycle_) {
269  param = lifecycle_node_->get_parameter(name);
270  } else {
271  param = node_->get_parameter(name);
272  }
273  updater(param);
274  } catch (const std::exception& e) {
275  auto logger = is_lifecycle_ ? lifecycle_node_->get_logger() : node_->get_logger();
276  RCLCPP_WARN(logger, "Failed to load bound parameter '%s': %s", name.c_str(), e.what());
277  }

285  void setupParameterCallback(std::function<bool(const std::vector<rclcpp::Parameter>&)> custom_callback = nullptr) {
286  auto callback = [this, custom_callback](const std::vector<rclcpp::Parameter>& params) -> rcl_interfaces::msg::SetParametersResult {
287  rcl_interfaces::msg::SetParametersResult result;
288  result.successful = true;
289 
290  auto logger = is_lifecycle_ ? lifecycle_node_->get_logger() : node_->get_logger();
291 
292  // First, update all bound parameters automatically
293  for (const auto& param : params) {
294  RCLCPP_DEBUG(logger, "Parameter '%s' changed to: %s", param.get_name().c_str(), param.value_to_string().c_str());
295 
296  // Check if this parameter has a binding
297  auto it = param_bindings_.find(param.get_name());
298  if (it != param_bindings_.end()) {
299  try {
300  it->second(param); // Execute the binding to update the variable
301  } catch (const std::exception& e) {
302  RCLCPP_ERROR(logger, "Failed to update bound parameter '%s': %s", param.get_name().c_str(), e.what());
303  result.successful = false;
304  result.reason = "Failed to update bound parameter: " + std::string(e.what());
305  return result;
306  }
307  }
308  }
309 
310  // If custom callback is provided, use it for additional validation
311  if (custom_callback) {
312  result.successful = custom_callback(params);
313  if (!result.successful) {
314  result.reason = "Custom parameter validation failed";
315  }
316  }
317 
318  return result;
319  };
320 
321  // Store the callback handle to keep it alive
322  if (is_lifecycle_) {
323  param_callback_handle_lifecycle_ = lifecycle_node_->add_on_set_parameters_callback(callback);
324  } else {
325  param_callback_handle_ = node_->add_on_set_parameters_callback(callback);
326  }
327  }

356  template <typename T>
357  bool setParam(const std::string& name, const T& value) {
358  try {
359  rcl_interfaces::msg::SetParametersResult result;
360  if (is_lifecycle_) {

362  } else {
363  result = node_->set_parameter(rclcpp::Parameter(name, value));
364  }

366  } catch (const std::exception& e) {
367  auto logger = is_lifecycle_ ? lifecycle_node_->get_logger() : node_->get_logger();
368  RCLCPP_ERROR(logger, "Failed to set parameter '%s': %s", name.c_str(), e.what());
369  return false;
370  }
