453 template <typename GraphType>
454 void TebVisualization::publishGraph(const GraphType& graph, const std::string& ns_prefix) {
455  if (printErrorWhenNotInitialized()) return;
456 
457  using GraphVertexIterator = typename boost::graph_traits<GraphType>::vertex_iterator;
458  using GraphEdgeIterator = typename boost::graph_traits<GraphType>::edge_iterator;
459 
460  // Visualize Edges
461  visualization_msgs::msg::Marker marker;
462  marker.header.frame_id = cfg_->map_frame;
463  marker.header.stamp = node_->now();
464  marker.ns = ns_prefix + "Edges";
465  marker.id = 0;
466 // #define TRIANGLE
467 #ifdef TRIANGLE
468  marker.type = visualization_msgs::msg::Marker::TRIANGLE_LIST;
469 #else
470  marker.type = visualization_msgs::msg::Marker::LINE_LIST;
471 #endif
472  marker.action = visualization_msgs::msg::Marker::ADD;
473 
474  GraphEdgeIterator it_edge;
475  GraphEdgeIterator end_edges;
476  for (boost::tie(it_edge, end_edges) = boost::edges(graph); it_edge != end_edges; ++it_edge) {
477 #ifdef TRIANGLE
478  geometry_msgs::msg::Point point_start1;
479  point_start1.x = graph[boost::source(*it_edge, graph)].pos[0] + 0.05;
480  point_start1.y = graph[boost::source(*it_edge, graph)].pos[1] - 0.05;
481  point_start1.z = 0;
482  marker.points.push_back(point_start1);
483  geometry_msgs::msg::Point point_start2;
484  point_start2.x = graph[boost::source(*it_edge, graph)].pos[0] - 0.05;
485  point_start2.y = graph[boost::source(*it_edge, graph)].pos[1] + 0.05;
486  point_start2.z = 0;
487  marker.points.push_back(point_start2);
488 
489 #else
490  geometry_msgs::msg::Point point_start;
491  point_start.x = graph[boost::source(*it_edge, graph)].pos[0];
492  point_start.y = graph[boost::source(*it_edge, graph)].pos[1];
493  point_start.z = 0;
494  marker.points.push_back(point_start);
495 #endif
496  geometry_msgs::msg::Point point_end;
497  point_end.x = graph[boost::target(*it_edge, graph)].pos[0];
498  point_end.y = graph[boost::target(*it_edge, graph)].pos[1];
499  point_end.z = 0;
500  marker.points.push_back(point_end);
501 
502  // add color
503  std_msgs::msg::ColorRGBA color;
504  color.a = 1.0;
505  color.r = 0;
506  color.g = 0;
507  color.b = 1;
508  marker.colors.push_back(color);
509  marker.colors.push_back(color);
510 #ifdef TRIANGLE
511  marker.colors.push_back(color);
512 #endif
513  }
514 
515 #ifdef TRIANGLE
516  marker.scale.x = 1;
517  marker.scale.y = 1;
518  marker.scale.z = 1;
519 #else
520  marker.scale.x = 0.01;
521 #endif
522  marker.color.a = 1.0;
523  marker.color.r = 0.0;
524  marker.color.g = 1.0;
525  marker.color.b = 0.0;
526 
527  // Now publish edge markers
528  teb_marker_pub_->publish(marker);
529 
530  // Visualize vertices
531  marker.header.frame_id = cfg_->map_frame;
532  marker.header.stamp = node_->now();
533  marker.ns = ns_prefix + "Vertices";
534  marker.id = 0;
535  marker.type = visualization_msgs::msg::Marker::POINTS;
536  marker.action = visualization_msgs::msg::Marker::ADD;
537 
538  GraphVertexIterator it_vert;
539  GraphVertexIterator end_vert;
540  for (boost::tie(it_vert, end_vert) = boost::vertices(graph); it_vert != end_vert; ++it_vert) {
541  geometry_msgs::msg::Point point;
542  point.x = graph[*it_vert].pos[0];
543  point.y = graph[*it_vert].pos[1];
544  point.z = 0;
545  marker.points.push_back(point);
546  // add color
547 
548  std_msgs::msg::ColorRGBA color;
549  color.a = 1.0;
550  if (it_vert == end_vert - 1) {
551  color.r = 1;
552  color.g = 0;
553  color.b = 0;
554  } else {
555  color.r = 0;
556  color.g = 1;
557  color.b = 0;
558  }
559  marker.colors.push_back(color);
560  }
561  // set first color (start vertix) to blue
562  if (!marker.colors.empty()) {
563  marker.colors.front().b = 1;
564  marker.colors.front().g = 0;
565  }
566 
567  marker.scale.x = 0.1;
568  marker.scale.y = 0.1;
569  marker.color.a = 1.0;
570  marker.color.r = 0.0;
571  marker.color.g = 1.0;
572  marker.color.b = 0.0;
573 
574  // Now publish vertex markers
575  teb_marker_pub_->publish(marker);
576 }

578 template <typename BidirIter>
579 void TebVisualization::publishPathContainer(BidirIter first, BidirIter last, const std::string& ns) {
580  if (printErrorWhenNotInitialized()) return;
581 
582  visualization_msgs::msg::Marker marker;
583  marker.header.frame_id = cfg_->map_frame;
584  marker.header.stamp = node_->now();
585  marker.ns = ns;
586  marker.id = 0;
587  marker.type = visualization_msgs::msg::Marker::LINE_LIST;
588  marker.action = visualization_msgs::msg::Marker::ADD;
589 
590  using PathType = typename std::iterator_traits<BidirIter>::value_type; // Get type of the path (point container)
591 
592  // Iterate through path container
593  while (first != last) {
594  // iterate single path points
595  typename PathType::const_iterator it_point;
596  typename PathType::const_iterator end_point;
597  for (it_point = first->begin(), end_point = boost::prior(first->end()); it_point != end_point; ++it_point) {
598  geometry_msgs::msg::Point point_start;
599  point_start.x = get_const_reference(*it_point).x();
600  point_start.y = get_const_reference(*it_point).y();
601  point_start.z = 0;
602  marker.points.push_back(point_start);
603 
604  geometry_msgs::msg::Point point_end;
605  point_end.x = get_const_reference(*boost::next(it_point)).x();
606  point_end.y = get_const_reference(*boost::next(it_point)).y();
607  point_end.z = 0;
608  marker.points.push_back(point_end);
609  }
610  ++first;
611  }
612  marker.scale.x = 0.01;
613  marker.color.a = 1.0;
614  marker.color.r = 0.0;
615  marker.color.g = 1.0;
616  marker.color.b = 0.0;
617 
618  teb_marker_pub_->publish(marker);
619 }
