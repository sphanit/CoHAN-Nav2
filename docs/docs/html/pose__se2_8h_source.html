255  void rotateGlobal(double angle, bool adjust_theta = true) {
256  double new_x = (std::cos(angle) * position_.x()) - (std::sin(angle) * position_.y());
257  double new_y = (std::sin(angle) * position_.x()) + (std::cos(angle) * position_.y());
258  position_.x() = new_x;
259  position_.y() = new_y;
260  if (adjust_theta) theta_ = g2o::normalize_theta(theta_ + angle);
261  }
