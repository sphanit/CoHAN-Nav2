67  void computeError() override {
68  HATEB_ASSERT_MSG(cfg_, "You must call setParameters() on EdgeStaticAgentVisibility()");
69  const auto* robot_bandpt = static_cast<const VertexPose*>(_vertices[0]);
70 
71  Eigen::Vector2d d_rtoh = static_agent_.position() - robot_bandpt->position();
72  Eigen::Vector2d d_htor = robot_bandpt->position() - static_agent_.position();
73  Eigen::Vector2d agent_look_at = {cos(static_agent_.theta()), sin(static_agent_.theta())};
74  Eigen::Vector2d robot_look_at = {cos(robot_bandpt->theta()), sin(robot_bandpt->theta())};
75 
76  // calculate the angle between human heading and direction vector from human to robot
77  double delta_psi = fabs(acos(agent_look_at.dot(d_htor) / (agent_look_at.norm() * d_htor.norm())));
78  double c_visibility = 0.0;
79  double ang = agent_look_at.dot(robot_look_at);
80 
81  // Apply the cost if the robot is not within the FOV of human (behind the human)
82  if (delta_psi >= cfg_->agent.fov * M_PI / 180) {
83  if (ang >= 0) {
84  c_visibility = 5 * ((std::pow(2, -(std::pow(d_rtoh.x(), 2)))) * (std::pow(2, -(std::pow(d_rtoh.y(), 2)))));
85  } else {
86  c_visibility = 0.;
87  }
88  }
89 
90  _error[0] = penaltyBoundFromAbove(c_visibility, cfg_->hateb.visibility_cost_threshold, cfg_->optim.penalty_epsilon);
91 
92  HATEB_ASSERT_MSG(std::isfinite(_error[0]), "EdgeStaticAgentVisibility::computeError() _error[0]=%f\n", _error[0]);
93  }
