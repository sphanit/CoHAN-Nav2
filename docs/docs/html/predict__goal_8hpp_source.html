142  void getProbabilities(int id) {
143  std::vector<double> probs; // Temporary storage for probabilities
144  std::vector<double> dists; // Storage for distances to goals
145 
146  if (goal_priors_[id].empty()) {
147  double prior = 1.0 / static_cast<double>(goals_.size());
148  for (int i = 0; i < goals_.size(); i++) {
149  goal_priors_[id].push_back(prior);
150  }
151  }
152 
153  auto trajectory = agents_trajs_[id];
154  int goal_id = 0;
155  double probs_sum = 0;
156  bool still = false;
157 
158  for (auto &goal : goals_) {
159  double probability = 1.0 / goals_.size();
160  int n = trajectory.size();
161  for (int i = 1; i < n; i++) {
162  Eigen::Vector2d heading = trajectory[i] - trajectory[i - 1];
163  if (heading.norm() == 0) {
164  still = true;
165  break;
166  }
167 
168  Eigen::Vector2d goal_vec = goal - trajectory[i];
169  if (i == n - 1) {
170  dists.push_back(goal_vec.norm());
171  }
172 
173  double phi = 0;
174  if (heading.norm() != 0) {
175  phi = std::acos(heading.dot(goal_vec) / (heading.norm() * goal_vec.norm()));
176  }
177 
178  double g = std::exp((i - n) / 0.5);
179  probability *= std::pow(nd_.pdf(phi), g); // Combined proability for the entire trajectory
180  }
181 
182  if (!still) {
183  double goal_probability = goal_priors_[id][goal_id];
184  goal_probability *= probability;
185  probs_sum += goal_probability;
186  probs.push_back(goal_probability);
187  }
188 
189  goal_id++;
190  }
191 
192  if (!still) {
193  for (auto &prob : probs) {
194  prob = prob / probs_sum;
195  double prior = prob;
196  }
197  agent_probs_[id] = probs;
198  }
199  }
